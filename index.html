import express from "express"; import dotenv from "dotenv"; import OpenAI from "openai"; import path from "path"; import { fileURLToPath } from "url"; dotenv.config(); const app = express(); const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY }); const __filename = fileURLToPath(import.meta.url); const __dirname = path.dirname(__filename); app.use(express.json()); app.use(express.static(__dirname)); app.post("/api/chat", async (req, res) => { try { const userMessage = req.body.message; const completion = await openai.chat.completions.create({ model: "gpt-4o-mini", messages: [ { role: "system", content: "Você é um assistente virtual integrado ao site A Cara do Pedro. Fale apenas em português do Brasil, de forma formal e objetiva." }, { role: "user", content: userMessage } ], max_tokens: 300 }); res.json({ reply: completion.choices[0].message.content }); } catch (error) { res.status(500).json({ reply: "Erro ao processar a solicitação." }); } }); app.listen(3000, () => { console.log("Servidor rodando em http://localhost:3000"); });
